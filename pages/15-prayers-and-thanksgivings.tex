\bcpchapter{Prayers and Thanksgivings}

\subsection{Prayers.}

\subsubsection{For Rain.}

\noindent
\lett{O}{ God}, heavenly Father, who by thy Son Jesus Christ hast promised to all those who seek thy kingdom, and the righteousness thereof, all things necessary to their bodily sustenance; Send us, we beseech thee, in this our necessity, such moderate rain and showers, that we may receive the fruits of the earth to our comfort, and to thy honour; through Jesus Christ our Lord.
\inlineresp{Amen.}
